#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <ctype.h>

#define MAX_CODE 5
#define MAX_NAME 50

// Function declarations
void mainMenu(void);
void generateCode(char secret[], int length);
int validateGuess(char guess[], int length);
void giveFeedback(char secret[], char guess[], int length);
void playGame(void);
void saveScore(char name[], int score, char difficulty[]);
void showLeaderboard(void);
int getNumber(int min, int max);
int chooseDifficulty(int *attempts, char difficulty[]);

int main(void)
{
    srand((unsigned int)time(NULL));
    mainMenu();
    return 0;
}

// Display the main menu
void mainMenu(void)
{
    int choice;

    do
    {
        printf("\n====================================\n");
        printf("          CODE BREAKER GAME\n");
        printf("====================================\n");
        printf("1. Start New Game\n");
        printf("2. View Leaderboard\n");
        printf("3. Exit Game\n");
        printf("====================================\n");
        printf("Enter your choice: ");

        choice = getNumber(1, 3);

        switch (choice)
        {
            case 1:
                playGame();
                break;
            case 2:
                showLeaderboard();
                break;
            case 3:
                printf("\nThank you for playing CODE BREAKER!\n");
                break;
        }
    } while (choice != 3);
}

// Read and validate an integer within a range
int getNumber(int min, int max)
{
    char input[100];
    char *end;
    long number;

    while (1)
    {
        if (fgets(input, sizeof(input), stdin) == NULL)
            return min;

        number = strtol(input, &end, 10);

        while (isspace((unsigned char)*end))
            end++;

        if (end != input && *end == '\0' &&
            number >= min && number <= max)
            return (int)number;

        printf("Invalid input! Enter a number between %d and %d: ",
               min, max);
    }
}

// Let the player select a difficulty level
int chooseDifficulty(int *attempts, char difficulty[])
{
    int choice;

    printf("\nSELECT DIFFICULTY\n");
    printf("1. Easy   (3-digit code, 10 attempts)\n");
    printf("2. Medium (4-digit code, 8 attempts)\n");
    printf("3. Hard   (5-digit code, 6 attempts)\n");
    printf("Choose difficulty: ");

    choice = getNumber(1, 3);

    if (choice == 1)
    {
        *attempts = 10;
        strcpy(difficulty, "Easy");
        return 3;
    }
    else if (choice == 2)
    {
        *attempts = 8;
        strcpy(difficulty, "Medium");
        return 4;
    }

    *attempts = 6;
    strcpy(difficulty, "Hard");
    return 5;
}

// Generate a random code with unique digits
void generateCode(char secret[], int length)
{
    int digits[10];
    int i, j, temp;

    for (i = 0; i < 10; i++)
        digits[i] = i;

    // Shuffle the digits
    for (i = 9; i > 0; i--)
    {
        j = rand() % (i + 1);
        temp = digits[i];
        digits[i] = digits[j];
        digits[j] = temp;
    }

    for (i = 0; i < length; i++)
        secret[i] = (char)('0' + digits[i]);

    secret[length] = '\0';
}

// Check that the guess has the correct length and unique digits
int validateGuess(char guess[], int length)
{
    int i, j;

    if (strlen(guess) != (size_t)length)
        return 0;

    for (i = 0; i < length; i++)
    {
        if (!isdigit((unsigned char)guess[i]))
            return 0;

        for (j = i + 1; j < length; j++)
        {
            if (guess[i] == guess[j])
                return 0;
        }
    }

    return 1;
}

// Compare the guess with the secret code and display feedback
void giveFeedback(char secret[], char guess[], int length)
{
    int exact = 0;
    int misplaced = 0;
    int i, j;

    for (i = 0; i < length; i++)
    {
        if (guess[i] == secret[i])
        {
            exact++;
        }
        else
        {
            for (j = 0; j < length; j++)
            {
                if (guess[i] == secret[j])
                {
                    misplaced++;
                    break;
                }
            }
        }
    }

    printf("\n---------- FEEDBACK ----------\n");
    printf("Exact matches: %d\n", exact);
    printf("Wrong position: %d\n", misplaced);
    printf("Not present: %d\n", length - exact - misplaced);
    printf("------------------------------\n");
}

// Save a winning player's score in leaderboard.txt
void saveScore(char name[], int score, char difficulty[])
{
    FILE *fp = fopen("leaderboard.txt", "a");

    if (fp == NULL)
    {
        printf("Unable to save score!\n");
        return;
    }

    fprintf(fp, "%s %d %s\n", name, score, difficulty);
    fclose(fp);
}

// Read and display saved leaderboard records
void showLeaderboard(void)
{
    FILE *fp;
    char name[MAX_NAME];
    char difficulty[20];
    int score;
    int found = 0;

    fp = fopen("leaderboard.txt", "r");

    printf("\n====================================\n");
    printf("             LEADERBOARD\n");
    printf("====================================\n");

    if (fp == NULL)
    {
        printf("No records found yet!\n");
        return;
    }

    printf("%-20s %-10s %-10s\n", "PLAYER", "SCORE", "LEVEL");
    printf("------------------------------------\n");

    while (fscanf(fp, "%49s %d %19s", name, &score, difficulty) == 3)
    {
        printf("%-20s %-10d %-10s\n", name, score, difficulty);
        found = 1;
    }

    if (!found)
        printf("No records found yet!\n");

    fclose(fp);
}

// Run one round of the game
void playGame(void)
{
    char secret[MAX_CODE + 1];
    char guess[100];
    char name[MAX_NAME];
    char difficulty[20];

    int length;
    int attempts;
    int remaining;
    int score = 0;
    int won = 0;
    int hintUsed = 0;
    int i;
    int hintPosition;

    printf("\nEnter your player name: ");

    if (fgets(name, sizeof(name), stdin) == NULL)
        return;

    name[strcspn(name, "\n")] = '\0';

    // Replace spaces so each name is stored as one leaderboard field
    for (i = 0; name[i] != '\0'; i++)
    {
        if (isspace((unsigned char)name[i]))
            name[i] = '_';
    }

    if (strlen(name) == 0)
        strcpy(name, "Player");

    length = chooseDifficulty(&attempts, difficulty);
    generateCode(secret, length);
    remaining = attempts;

    printf("\n====================================\n");
    printf("             GAME STARTED!\n");
    printf("====================================\n");
    printf("Difficulty: %s\n", difficulty);
    printf("Code length: %d digits\n", length);
    printf("Total attempts: %d\n", attempts);

    printf("\nRULES:\n");
    printf("1. Guess the secret code.\n");
    printf("2. Digits cannot be repeated.\n");
    printf("3. Feedback shows exact and misplaced digits.\n");
    printf("4. You can request one hint.\n");

    while (remaining > 0)
    {
        printf("\nAttempts remaining: %d\n", remaining);
        printf("Enter your %d-digit guess: ", length);

        if (fgets(guess, sizeof(guess), stdin) == NULL)
            return;

        guess[strcspn(guess, "\n")] = '\0';

        if (!validateGuess(guess, length))
        {
            printf("Invalid guess! Enter exactly %d unique digits.\n",
                   length);
            continue;
        }

        if (strcmp(secret, guess) == 0)
        {
            won = 1;
            break;
        }

        giveFeedback(secret, guess, length);
        remaining--;

        if (remaining > 0 && !hintUsed)
        {
            char choice[20];

            printf("\nDo you want a hint? (y/n): ");

            if (fgets(choice, sizeof(choice), stdin) == NULL)
                return;

            if (choice[0] == 'y' || choice[0] == 'Y')
            {
                hintPosition = rand() % length;
                printf("HINT: Position %d contains digit %c\n",
                       hintPosition + 1, secret[hintPosition]);
                hintUsed = 1;
            }
        }
    }

    printf("\n====================================\n");

    if (won)
    {
        printf("          CONGRATULATIONS!\n");
        printf("             CODE CRACKED!\n");

        score = remaining * 10;

        if (strcmp(difficulty, "Easy") == 0)
            score += 10;
        else if (strcmp(difficulty, "Medium") == 0)
            score += 20;
        else
            score += 30;

        if (hintUsed)
            score -= 5;

        if (score < 0)
            score = 0;

        printf("Player: %s\n", name);
        printf("Secret code: %s\n", secret);
        printf("Final score: %d\n", score);

        saveScore(name, score, difficulty);
    }
    else
    {
        printf("               GAME OVER!\n");
        printf("You failed to crack the code.\n");
        printf("Secret code was: %s\n", secret);
    }

    printf("====================================\n");
}
